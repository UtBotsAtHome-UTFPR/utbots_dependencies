cmake_minimum_required(VERSION 3.5)
project(utbots_actions)

# Default to C99
if(NOT CMAKE_C_STANDARD)
  set(CMAKE_C_STANDARD 99)
endif()

# Default to C++14
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 14)
endif()

if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# Find dependencies
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(utbots_msgs REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "action/Extract3DPoint.action"
  "action/Recognition.action"
  "action/NewFace.action"
  "action/Train.action"
  "action/YOLODetection.action"
  "action/MPPose.action"
  "action/InterpretNLU.action"
  DEPENDENCIES std_msgs geometry_msgs sensor_msgs utbots_msgs
)

ament_export_dependencies(rosidl_default_runtime)

if(BUILD_TESTING)
  find_package(ament_lint_auto REQUIRED)
  ament_lint_auto_find_test_dependencies()
endif()

ament_package()
